## 总则

### 1. 执行纪律

1. **渐进式重构**：Phase 0 拆分 main.py 时，每拆一个模块跑一次测试，保证 26 个测试始终通过。禁止"大爆炸式"一次性重写。
2. **不破坏旧功能**：v2 的单章手动模式必须保留为 fallback，九步管线是新增路径不是替换。
3. **提示词与代码分离**：所有 prompt 进 YAML，代码只负责加载和渲染。
4. **做减法优先**：deepen_and_bury 的字数应 ≤ 初稿 × 1.05，禁止大幅加字。
5. **每 Phase 交付可运行产物**：不积累半成品。

### 2. 代码规范

- Python：类型注解必填，Pydantic 模型，函数 docstring
- TypeScript：严格模式，接口定义先行
- 命名：workflows 用动词原形（generate/archaeology/deepen），表名用复数名词
- 错误处理：工作流内部 catch 异常并返回结构化结果，不向上抛裸异常

### 1.4 Orchestrator 调度器

```python
# engine/orchestrator.py
class Orchestrator:
    def run(self):
        job = get_job(self.job_id)
        blueprint = get_blueprint(job.blueprint_id)
        
        for offset in range(job.target_chapter_count):
            chapter_number = job.start_chapter_number + offset
            
            self._wait_if_paused()
            self._wait_if_checkpoint_pending()
            
            try:
                chapter = create_chapter(job.project_id, chapter_number)
                
                # 装配上下文（含叙事记忆注入）
                ctx = ChapterContext(
                    project_id=job.project_id,
                    chapter_id=chapter.id,
                    chapter_number=chapter_number,
                    blueprint=blueprint,
                    target_words=blueprint.params.words_per_chapter,
                    narrative_memory=get_active_narrative_memory(job.project_id),
                    prev_bridge=get_previous_bridge(job.project_id, chapter_number),
                )
                
                self.pipeline.run(ctx, job.params, on_step=self._on_step)
                
                # 考古发现写入叙事记忆（v4 关键：横向反馈）
                self._harvest_narrative_memory(ctx)
                
                # 智能停检测（v4 补充情感节奏条件）
                should_stop, reason = self.checkpoint.should_smart_stop(ctx)
                if should_stop:
                    self._pause_with_reason("smart_stop", reason, chapter)
                    self._wait_if_checkpoint_pending()
                
                elif self.checkpoint.hit_checkpoint(job.checkpoint_strategy, offset):
                    self._pause_with_reason("checkpoint", "", chapter)
                    self._wait_if_checkpoint_pending()
                
                if job.auto_finalize:
                    finalize_chapter(job.project_id, chapter.id)
                
                update_job_progress(self.job_id, chapter_number)
                
            except Exception as e:
                self.breaker.record_failure(e)
                if self.breaker.should_trip():
                    self._fail_job(f"熔断：连续失败 {self.breaker.consecutive_failures} 次")
                    return
                record_chapter_failure(self.job_id, chapter_number, str(e))
        
        complete_job(self.job_id)
    
    def _harvest_narrative_memory(self, ctx: ChapterContext):
        """v4 关键：把考古发现写入叙事记忆，注入下一章种子。"""
        archaeology = ctx.archaeology
        if not archaeology:
            return
        
        # 提取有价值的线索
        leads = extract_narrative_leads(archaeology)
        for lead in leads:
            insert_narrative_memory(
                project_id=ctx.project_id,
                source_chapter_id=ctx.chapter_id,
                memory_type=lead.type,
                memory_content=lead.content,
            )
```

#### 2.2.2 workflows/dialogue.py（对话潜台词挖掘，新增）

```python
class DialogueSubtextWorkflow:
    """识别对话段落，分析潜台词。"""
    
    def run(self, chapter_id: str, content: str, characters: list) -> dict:
        # 1. 识别所有对话段落（引号包围的文本 + 上下文）
        dialogues = self._extract_dialogues(content)
        # 2. 逐段分析（调用 LLM）
        analyses = []
        for d in dialogues:
            analysis = self._analyze_subtext(d, characters)
            analyses.append({
                "position": d.position,
                "speaker": d.speaker,
                "line": d.text,
                "issues": analysis.issues,  # said_subtext / missing_tone / voice_mismatch / rhythm_flat
                "subtext": analysis.subtext,
                "emotional_debt": analysis.debt,  # 角色未表达的情感
                "fix_direction": analysis.fix,
            })
        return {"dialogue_map": analyses}
```

问题类型枚举：
- `said_subtext`：潜台词被说破（需藏回）
- `missing_tone`：缺少语气层（需补动作/神态）
- `voice_mismatch`：声纹不一致（需对照声纹库修正）
- `rhythm_flat`：节奏单一（可加沉默/停顿）
- `debt_accumulation`：情感债务积压（追踪不修改）

```python
class DeepenAndBuryWorkflow:
    """融合三份产出（对话地图+隐藏层地图+追读力报告），统一做减法。"""
    
    def run(self, chapter_id: str, content: str, 
            dialogue_map: dict, archaeology: dict, reader_pull: dict) -> dict:
        # 1. 合并所有修改建议
        opportunities = self._merge_opportunities(
            dialogue_map, archaeology, reader_pull
        )
        # 2. 调用 LLM 做加深（prompt 强调做减法）
        revised = self._deepen(content, opportunities)
        # 3. 字数检查（不应大幅增加）
        if len(revised) > len(content) * 1.05:
            # 触发告警，但不阻断
            log.warn("deepen 增加字数过多，可能未遵循做减法原则")
        return {
            "revised_text": revised,
            "changes": [...],  # 每处修改的说明
        }
```

```python
        return {"tracked_images": [...]}
```

```python
def check_foreshadowing_plan(blueprint, chapter_number):
    """当前章号 == planted_in → brief 提示埋伏笔
       当前章号 == payoff_in → brief 提示回收伏笔"""
    to_plant = [f for f in blueprint.foreshadowings

    to_payoff = [f for f in blueprint.foreshadowings 
                 if f.payoff_in == chapter_number and f.status == "planted"]
```

```python
# engine/orchestrator.py
def resume_from_checkpoint(job_id):
    """任务恢复时，检查已完成的可信步骤，不重跑。"""
    job = get_job(job_id)
    for chapter_number in range(job.start_chapter_number, job.current_chapter_number):
        steps = get_chapter_steps(job_id, chapter_number)
        # 如果该章所有必需步骤都 completed，跳过
        if all(s.status == "completed" for s in steps if s.required):
            continue
        # 否则从第一个未完成步骤继续
        resume_chapter_from_step(chapter_number, first_incomplete_step(steps))
```
